## 一、设计目标

在 Ling-iflow 项目中新增飞书渠道支持，实现与企业微信渠道一致的业务逻辑，同时保持架构统一性和代码复用性。

### 1.1 核心原则

- **架构一致**: 复用现有消息总线架构，与企业微信渠道保持一致
- **业务统一**: 消息内容、命令系统、通知机制与企业微信渠道完全一致
- **参考最佳实践**: 连接方式和功能范围参考 iflow-bot 项目实现
- **配置简化**: 复用现有 bots 配置结构，降低用户配置成本

## 二、架构决策

### 2.1 连接方式：WebSocket 长连接 ⭐

**决策**: 采用 WebSocket 长连接方式（与 iflow-bot 一致）

| 对比项 | HTTP 回调 | WebSocket 长连接 |
|-------|----------|-----------------|
| 公网 IP | 需要 | **无需** |
| SDK | 原生 httpx | lark-oapi 官方 SDK |
| 代码量 | 400-500 行 | 1000+ 行 |
| 功能 | 基础文本 | 完整功能 |
| 流式输出 | ❌ | ✅ |
| 文件传输 | ❌ | ✅ |
| 打字动画 | ❌ | ✅ |

**优势**:
- 无需公网 IP，降低部署门槛
- 使用官方 SDK，稳定性和兼容性好
- 支持流式输出、文件传输等高级功能

### 2.2 功能范围：完整功能 ⭐

**决策**: 实现完整功能（与 iflow-bot 一致）

**基础功能**:

- ✅ 白名单控制

async def _on_start(self) -> None:
    """启动时的钩子方法"""
    logger.info(f"[{self.module_id}] 启动飞书通道")
    
    if not FEISHU_AVAILABLE:
        logger.error(f"[{self.module_id}] lark-oapi not installed")

    # 初始化 SDK 客户端
    self._client = lark.Client.builder() \
        .app_id(self.app_id) \
        .app_secret(self.app_secret) \
        .log_level(logging.INFO) \
        .build()
    
    # 配置 WebSocket 客户端
    self._ws_client = lark.ws.Client.builder() \
        .app_id(self.app_id) \
        .app_secret(self.app_secret) \
        .event_handler(self._handle_event) \
        .build()
    
    # 启动 WebSocket 线程
    self._start_websocket_thread()
    
    # 等待 WebSocket 连接建立
    await asyncio.sleep(2)
    
    self._running = True
    logger.info(f"[{self.module_id}] 飞书通道启动完成（WebSocket 长连接）")
    
    # 发送启动通知
    await self._send_startup_notification()

#### 5.2.5 消息接收方法

```python
async def _on_receive(self, message: Message) -> None:
    """接收消息时的钩子方法"""
    try:
        if not self._client:
            logger.error(f"[{self.module_id}] 客户端未初始化")
            return
        
        # 转换为飞书消息格式
        platform_msg = self.converter.to_platform_message("feishu", message)
        
        # 提取消息内容
        content = platform_msg.get("content", "")
        receive_id = platform_msg.get("receive_id")
        msg_type = platform_msg.get("msg_type", "text")
        
        # 确定 receive_id_type
        receive_id_type = "chat_id" if receive_id.startswith("oc_") else "open_id"
        
        # 发送到飞书
        success = await self.send_message(
            content=content,
            receive_id=receive_id,
            receive_id_type=receive_id_type,
            msg_type=msg_type
        )
        
        if success:
            logger.info(f"[{self.module_id}] 消息已发送到飞书")
        else:
            logger.error(f"[{self.module_id}] 消息发送失败")
            
    except Exception as e:
        logger.error(f"[{self.module_id}] 处理消息失败：{e}", exc_info=True)
```

#### 5.2.6 消息发送方法

```python
async def send_message(
    self,
    content: str,
    receive_id: str,
    receive_id_type: str,
    msg_type: str = "text"
) -> bool:
    """发送消息到飞书"""
    if not self._client:
        logger.error(f"[{self.module_id}] 客户端未初始化")
        return False
    
    try:
        loop = asyncio.get_running_loop()
        
        # 根据消息类型构建内容
        if msg_type == "text":
            # 转换为交互卡片（支持 Markdown）
            card_content = json.dumps({
                "config": {"wide_screen_mode": True},
                "elements": self._build_card_elements(content)
            }, ensure_ascii=False)
            
            # 同步发送
            message_id = await loop.run_in_executor(
                None,
                self._send_message_sync,
                receive_id_type,
                receive_id,
                "interactive",
                card_content
            )
        else:
            # 其他消息类型
            message_id = await loop.run_in_executor(
                None,
                self._send_message_sync,
                receive_id_type,
                receive_id,
                msg_type,
                content
            )
        
        return message_id is not None
        
    except Exception as e:
        logger.error(f"[{self.module_id}] 发送消息异常：{e}", exc_info=True)
        return False
```

#### 5.2.8 消息解析方法

```python
async def _handle_message_event(self, event: P2ImMessageReceiveV1) -> None:
    """处理消息接收事件"""
    try:
        message = event.event.message
        message_id = message.message_id
        chat_id = message.chat_id
        sender_id = message.sender_id
        msg_type = message.message_type
        content = message.content
        
        # 消息去重
        if await self._is_duplicate_message(message_id):
            logger.info(f"[{self.module_id}] 忽略重复消息：{message_id}")
            return
        
        # 解析消息内容
        content_parts = []
        media_paths = []
        
        try:
            content_json = json.loads(content) if content else {}
        except json.JSONDecodeError:
            content_json = {}
        
        if msg_type == "text":
            text = content_json.get("text", "")
            if text:
                content_parts.append(text)
        
        elif msg_type == "post":
            # 解析富文本消息
            text_parts, resources = self._extract_post_parts(content_json)
            content_parts.extend(text_parts)
            # 下载媒体文件
            for resource in resources:
                file_path, content_text = await self._download_and_save_media(
                    resource, message_id
                )
                if file_path:
                    media_paths.append(file_path)
                if content_text:
                    content_parts.append(content_text)
        
        elif msg_type in ("image", "file", "audio", "media"):
            # 下载媒体文件
            file_path, content_text = await self._download_and_save_media(
                content_json, message_id, msg_type
            )
            if file_path:
                media_paths.append(file_path)
            content_parts.append(content_text)
        
        # 合并内容
        content = "\n".join(content_parts) if content_parts else ""
        
        if not content and not media_paths:
            return
        
        # 确定回复目标
        chat_type = "group" if chat_id.startswith("oc_") else "private"
        reply_to = chat_id if chat_type == "group" else sender_id
        
        # 创建标准消息
        message_obj = self.converter.from_platform_message(
            "feishu",
            {
                "content": content,
                "sender_id": sender_id,
                "chat_id": reply_to,
                "msg_type": msg_type
            },
            self.module_id
        )
        
        # 添加到消息总线
        if self.message_bus:
            await self.send_to_bus(message_obj)
            logger.info(f"[{self.module_id}] 飞书消息已发送到消息总线")
        
    except Exception as e:
        logger.error(f"[{self.module_id}] 处理消息失败：{e}", exc_info=True)
```

async def _create_feishu_modules(self) -> None:
    """创建所有飞书渠道模块"""
    bots_config = self.config.get("bots", [])
    feishu_configs = [b for b in bots_config if b.get("platform") == "feishu"]
    
    logger.info(f"创建 {len(feishu_configs)} 个飞书模块...")
    
    for fs_config in feishu_configs:
        if not fs_config.get("enabled", True):
            continue
            
        module_id = f"feishu_{fs_config['name']}"
        try:
            feishu = await self.manager.create_module(
                module_type="feishu",
                module_id=module_id,
                config=fs_config
            )
            feishu.set_manager(self.manager)
            feishu.set_bot(self)
            self.feishu_modules[fs_config['name']] = feishu
            await self.manager.register_module(module_id)
            logger.info(f"创建飞书模块：{module_id}")
        except Exception as e:
            logger.error(f"创建飞书模块失败 {fs_config['name']}: {e}")

def create_feishu_inbound_message(
    self,
    feishu_msg: Dict[str, Any],
    module_id: str
) -> Message:
    """创建飞书入站消息"""
    return Message(
        id=Message.generate_id(),
        type=MessageType.TEXT,
        content=MessageContent(type=MessageType.TEXT, text=feishu_msg["content"]),
        from_module=module_id,
        chat_id=feishu_msg.get("chat_id", ""),
        user_id=feishu_msg.get("sender_id", ""),
        metadata={"msg_type": feishu_msg.get("msg_type", "text")}
    )

def create_feishu_outbound_message(
    self,
    message: Message
) -> Dict[str, Any]:
    """创建飞书出站消息"""
    return {
        "content": message.content.text if message.content else "",
        "receive_id": message.chat_id or message.metadata.get("chat_id", ""),
        "msg_type": "text"
    }
