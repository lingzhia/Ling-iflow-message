## 概述

本文档描述 Ling-iflow 项目的扩展性设计和未来改进方向。

## 扩展性设计

### 1. 渠道扩展

**设计思路**：
- 定义 BaseChannel 抽象基类
- 统一消息处理接口
- 支持动态加载渠道

**BaseChannel 接口**：

```python
from abc import ABC, abstractmethod
from typing import Callable, Optional, Dict, Any

class BaseChannel(ABC):
    """渠道基类。"""
    
    name: str = "base"
    
    @abstractmethod
    async def start(self) -> bool:
        """启动渠道。"""
        pass
    
    @abstractmethod
    async def stop(self) -> None:
        """停止渠道。"""
        pass
    
    @abstractmethod
    async def send(
        self,
        content: str,
        chat_id: str,
        is_group: bool = False,
        **kwargs
    ) -> bool:
        """发送消息。
        
        Args:
            content: 消息内容
            chat_id: 聊天 ID
            is_group: 是否为群聊
            **kwargs: 其他参数
            
        Returns:
            是否发送成功
        """
        pass
    
    @abstractmethod
    def set_message_handler(self, handler: Callable) -> None:
        """设置消息处理器。
        
        Args:
            handler: 消息处理器函数
        """
        pass
    
    @abstractmethod
    def get_config_schema(self) -> Dict[str, Any]:
        """获取配置 Schema。"""
        pass
```

**扩展示例 - Telegram 渠道**：

```python
import httpx

class TelegramChannel(BaseChannel):
    """Telegram 渠道。"""
    
    name = "telegram"
    
    def __init__(self, config: Dict[str, Any]):
        self.token = config.get("token", "")
        self.webhook_url = config.get("webhook_url", "")
        self.message_handler = None
    
    async def start(self) -> bool:
        """启动渠道。"""
        # 设置 Webhook
        url = f"https://api.telegram.org/bot{self.token}/setWebhook"
        payload = {"url": self.webhook_url}
        
        async with httpx.AsyncClient() as client:
            response = await client.post(url, json=payload)
            result = response.json()
            return result.get("ok", False)
    
    async def stop(self) -> None:
        """停止渠道。"""
        # 删除 Webhook
        url = f"https://api.telegram.org/bot{self.token}/deleteWebhook"
        
        async with httpx.AsyncClient() as client:
            await client.post(url)
    
    async def send(
        self,
        content: str,
        chat_id: str,
        is_group: bool = False,
        **kwargs
    ) -> bool:
        """发送消息。"""
        url = f"https://api.telegram.org/bot{self.token}/sendMessage"
        payload = {
            "chat_id": chat_id,
            "text": content
        }
        
        async with httpx.AsyncClient() as client:
            response = await client.post(url, json=payload)
            result = response.json()
            return result.get("ok", False)
    
    def set_message_handler(self, handler: Callable) -> None:
        """设置消息处理器。"""
        self.message_handler = handler
    
    def get_config_schema(self) -> Dict[str, Any]:
        """获取配置 Schema。"""
        return {
            "token": {"type": "string", "required": True},
            "webhook_url": {"type": "string", "required": True}
        }
```

**渠道注册机制**：

```python
from typing import Dict, Type

class ChannelRegistry:
    """渠道注册表。"""
    
    _channels: Dict[str, Type[BaseChannel]] = {}
    
    @classmethod
    def register(cls, channel_class: Type[BaseChannel]) -> Type[BaseChannel]:
        """注册渠道。"""
        cls._channels[channel_class.name] = channel_class
        return channel_class
    
    @classmethod
    def get(cls, name: str) -> Optional[Type[BaseChannel]]:
        """获取渠道类。"""
        return cls._channels.get(name)
    
    @classmethod
    def list(cls) -> list:
        """列出所有渠道。"""
        return list(cls._channels.keys())

# 使用装饰器注册渠道
@ChannelRegistry.register
class TelegramChannel(BaseChannel):
    name = "telegram"
    # ...

# 创建渠道实例
channel_class = ChannelRegistry.get("telegram")
if channel_class:
    channel = channel_class(config)
```

### 2. 命令扩展

**设计思路**：
- 支持自定义命令前缀
- 支持命令注册机制
- 支持命令别名

**命令注册机制**：

```python
from typing import Callable, Dict, List

class CommandRegistry:
    """命令注册表。"""
    
    def __init__(self):
        self._commands: Dict[str, Callable] = {}
        self._aliases: Dict[str, str] = {}
        self._help: Dict[str, str] = {}
    
    def register(
        self,
        name: str,
        handler: Callable,
        aliases: List[str] = None,
        help_text: str = ""
    ) -> None:
        """注册命令。"""
        self._commands[name] = handler
        self._help[name] = help_text
        
        if aliases:
            for alias in aliases:
                self._aliases[alias] = name
    
    def get_handler(self, name: str) -> Optional[Callable]:
        """获取命令处理器。"""
        # 检查别名
        if name in self._aliases:
            name = self._aliases[name]
        
        return self._commands.get(name)
    
    def list_commands(self) -> List[str]:
        """列出所有命令。"""
        return list(self._commands.keys())
    
    def get_help(self, name: str) -> str:
        """获取命令帮助。"""
        return self._help.get(name, "")

# 使用示例
command_registry = CommandRegistry()

async def custom_command_handler(chat_id: str, args: List[str]) -> str:
    """自定义命令处理器。"""
    return "这是自定义命令的回复"

# 注册命令
command_registry.register(
    name="custom",
    handler=custom_command_handler,
    aliases=["c", "cust"],
    help_text="自定义命令示例"
)

# 调用命令
handler = command_registry.get_handler("custom")
if handler:
    result = await handler(chat_id, args)
```

### 3. 消息处理器扩展

**设计思路**：
- 支持多个消息处理器
- 支持处理器优先级
- 支持处理器链

**消息处理器接口**：

```python
from typing import Callable, Optional, Dict, Any
from abc import ABC, abstractmethod

class MessageHandler(ABC):
    """消息处理器基类。"""
    
    priority: int = 0
    
    @abstractmethod
    async def handle(
        self,
        content: str,
        sender_id: str,
        chat_id: str,
        is_group: bool,
        metadata: Optional[Dict[str, Any]] = None
    ) -> Optional[str]:
        """处理消息。
        
        Returns:
            处理结果，如果返回 None 则继续处理下一个处理器
        """
        pass
```

**消息处理器链**：

```python
class MessageProcessor:
    """消息处理器。"""
    
    def __init__(self):
        self._handlers: List[MessageHandler] = []
    
    def add_handler(self, handler: MessageHandler) -> None:
        """添加处理器。"""
        self._handlers.append(handler)
        # 按优先级排序
        self._handlers.sort(key=lambda h: h.priority, reverse=True)
    
    async def process(
        self,
        content: str,
        sender_id: str,
        chat_id: str,
        is_group: bool,
        metadata: Optional[Dict[str, Any]] = None
    ) -> str:
        """处理消息。"""
        for handler in self._handlers:
            try:
                result = await handler.handle(
                    content,
                    sender_id,
                    chat_id,
                    is_group,
                    metadata
                )
                
                if result is not None:
                    return result
            except Exception as e:
                logger.error(f"Handler error: {e}", exc_info=True)
        
        return ""

# 使用示例
class KeywordHandler(MessageHandler):
    """关键词处理器。"""
    
    priority = 10
    
    def __init__(self, keywords: Dict[str, str]):
        self.keywords = keywords
    
    async def handle(self, content: str, sender_id: str, chat_id: str, is_group: bool, metadata=None) -> Optional[str]:
        """处理关键词。"""
        for keyword, response in self.keywords.items():
            if keyword in content:
                return response
        
        return None

# 创建处理器
processor = MessageProcessor()

# 添加处理器
processor.add_handler(KeywordHandler({
    "帮助": "这是帮助信息",
    "你好": "你好！有什么我可以帮助你的吗？"
}))
```
